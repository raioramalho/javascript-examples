generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

model Produtos {
  id      Int    @id @default(autoincrement())
  produto String
  estoque Int

  estoques Estoque[]
}

model Estoque {
  id         Int      @id @default(autoincrement())
  produto    Int
  estoque    Int
  lote       String
  vencimento DateTime

  produtos Produtos @relation(fields: [produto], references: [id])
}

// id 2 produto 1 estoque 15 lote 2023B vencimento 19/08/2023
